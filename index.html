Google Apps Script Backend Setup

  1. Open a new Google Sheet for Sri Lakshmi Balaji School Attendance.
  2. Click Extensions → Apps Script and paste the code below.
  3. Click Deploy → New deployment → Web app (Execute as: Me, Who has access: Anyone).
  4. Copy the generated Web App URL and paste it into the input box on the Reports tab.
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Date", "Time", "Roll No", "Student Name", "Class", "Status", "Method"]);
  }
  var payload = JSON.parse(e.postData.contents);
  var records = payload.records || [];
  records.forEach(function(r) {
    sheet.appendRow([r.date, r.time, r.rollNo, r.name, r.className, r.status, r.method]);
  });
  return ContentService.createTextOutput(JSON.stringify({ status: "success", count: records.length }))
    .setMimeType(ContentService.MimeType.JSON);
}
